using UnityEngine; using UnityEngine.UI; // DAN SOLOMON - OPEN WORLD GAME PROTOTYPE // Unity C# public class OpenWorldGame : MonoBehaviour { [Header("PLAYER")] public CharacterController player; public Transform playerTransform; public float walkSpeed = 5f; public float runSpeed = 9f; public float jumpHeight = 1.5f; public float gravity = -20f; [Header("CAMERA")] public Camera gameCamera; public Vector3 cameraOffset = new Vector3(0, 4, -7); public float cameraSpeed = 3f; [Header("CAR")] public Rigidbody car; public Transform carTransform; public Transform carExitPoint; public float carAcceleration = 25f; public float carMaxSpeed = 35f; public float steeringSpeed = 70f; [Header("GAME")] public float maxHealth = 100f; public float health = 100f; public int money = 500; [Range(0, 5)] public int wantedLevel = 0; [Header("MISSION")] public Transform missionPoint; public float missionDistance = 3f; public int missionReward = 1000; [Header("UI")] public Text healthText; public Text moneyText; public Text wantedText; public Text missionText; private Vector3 velocity; private bool driving = false; private bool nearCar = false; private bool missionCompleted = false; private float yaw = 0f; private float pitch = 15f; void Start() { health = maxHealth; if (gameCamera == null) gameCamera = Camera.main; UpdateUI(); } void Update() { if (!driving) { PlayerMovement(); CameraMovement(); CheckCar(); CheckMission(); } else { CameraFollowCar(); if (Input.GetKeyDown(KeyCode.E)) ExitCar(); } UpdateUI(); } void FixedUpdate() { if (driving) DriveCar(); } // PLAYER MOVEMENT void PlayerMovement() { if (player == null) return; float horizontal = Input.GetAxis("Horizontal"); float vertical = Input.GetAxis("Vertical"); Vector3 direction = new Vector3(horizontal, 0f, vertical).normalized; if (direction.magnitude >= 0.1f) { float cameraY = gameCamera.transform.eulerAngles.y; float targetAngle = Mathf.Atan2(direction.x, direction.z) * Mathf.Rad2Deg + cameraY; Quaternion rotation = Quaternion.Euler(0f, targetAngle, 0f); playerTransform.rotation = Quaternion.Slerp( playerTransform.rotation, rotation, 10f * Time.deltaTime ); Vector3 moveDirection = rotation * Vector3.forward; float speed = Input.GetKey(KeyCode.LeftShift) ? runSpeed : walkSpeed; player.Move( moveDirection.normalized * speed * Time.deltaTime ); } if (player.isGrounded && velocity.y < 0) velocity.y = -2f; if ( Input.GetButtonDown("Jump") && player.isGrounded ) { velocity.y = Mathf.Sqrt( jumpHeight * -2f * gravity ); } velocity.y += gravity * Time.deltaTime; player.Move( velocity * Time.deltaTime ); } // THIRD PERSON CAMERA void CameraMovement() { if (gameCamera == null || playerTransform == null) return; yaw += Input.GetAxis("Mouse X") * cameraSpeed; pitch -= Input.GetAxis("Mouse Y") * cameraSpeed; pitch = Mathf.Clamp( pitch, -10f, 60f ); Quaternion rotation = Quaternion.Euler( pitch, yaw, 0f ); gameCamera.transform.position = playerTransform.position + rotation * cameraOffset; gameCamera.transform.LookAt( playerTransform.position + Vector3.up * 1.5f ); } // CHECK IF PLAYER IS CLOSE TO CAR void CheckCar() { if ( playerTransform == null || carTransform == null ) return; float distance = Vector3.Distance( playerTransform.position, carTransform.position ); nearCar = distance < 3f; if ( nearCar && Input.GetKeyDown(KeyCode.E) ) { EnterCar(); } } // ENTER VEHICLE void EnterCar() { driving = true; if (player != null) player.enabled = false; if (playerTransform != null) playerTransform.gameObject.SetActive(false); } // EXIT VEHICLE void ExitCar() { driving = false; if (playerTransform != null) { playerTransform.gameObject.SetActive(true); if (carExitPoint != null) { playerTransform.position = carExitPoint.position; } else if (carTransform != null) { playerTransform.position = carTransform.position + carTransform.right * 2f; } } if (player != null) player.enabled = true; } // VEHICLE DRIVING void DriveCar() { if (car == null) return; float acceleration = Input.GetAxis("Vertical"); float steering = Input.GetAxis("Horizontal"); if ( car.linearVelocity.magnitude < carMaxSpeed ) { car.AddForce( carTransform.forward * acceleration * carAcceleration, ForceMode.Acceleration ); } float speedFactor = Mathf.Clamp01( car.linearVelocity.magnitude / 5f ); carTransform.Rotate( 0f, steering * steeringSpeed * speedFactor * Time.fixedDeltaTime, 0f ); } // CAMERA WHILE DRIVING void CameraFollowCar() { if ( gameCamera == null || carTransform == null ) return; Vector3 targetPosition = carTransform.position - carTransform.forward * 8f + Vector3.up * 4f; gameCamera.transform.position = Vector3.Lerp( gameCamera.transform.position, targetPosition, 5f * Time.deltaTime ); gameCamera.transform.LookAt( carTransform.position + Vector3.up ); } // PLAYER DAMAGE public void TakeDamage(float damage) { health -= damage; if (health <= 0) Respawn(); } // PLAYER RESPAWN void Respawn() { health = maxHealth; if (playerTransform != null) { playerTransform.position = new Vector3( 0f, 2f, 0f ); } wantedLevel = 0; } // ADD MONEY public void AddMoney(int amount) { money += amount; } // SPEND MONEY public bool SpendMoney(int amount) { if (money < amount) return false; money -= amount; return true; } // WANTED LEVEL public void AddWantedLevel(int amount) { wantedLevel += amount; wantedLevel = Mathf.Clamp( wantedLevel, 0, 5 ); } public void RemoveWantedLevel() { wantedLevel = 0; } // MISSION SYSTEM void CheckMission() { if ( missionCompleted || missionPoint == null || playerTransform == null ) return; float distance = Vector3.Distance( playerTransform.position, missionPoint.position ); if (distance <= missionDistance) { missionCompleted = true; AddMoney( missionReward ); Debug.Log( "MISSION COMPLETE! +$" + missionReward ); } } // UI void UpdateUI() { if (healthText != null) { healthText.text = "HEALTH: " + Mathf.RoundToInt(health); } if (moneyText != null) { moneyText.text = "$" + money; } if (wantedText != null) { wantedText.text = "WANTED: " + new string( '★', wantedLevel ); } if (missionText != null) { missionText.text = missionCompleted ? "MISSION COMPLETE" : "GO TO THE MISSION POINT"; } } }
