import express from "express"; import multer from "multer"; import cors from "cors"; import RunwayML, { toFile } from "@runwayml/sdk"; const app = express(); const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: 200 * 1024 * 1024 } });
