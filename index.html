//+------------------------------------------------------------------+ //| DAN_GOLD_EA_V2.mq5 | //| Test on DEMO first | //+------------------------------------------------------------------+ #property strict #property version "2.00" #include CTrade trade; input ulong MagicNumber = 260101; input ENUM_TIMEFRAMES SignalTF = PERIOD_M5; // RISK input double RiskPercent = 0.50; input double DailyLossLimitPct = 2.00; // SIGNAL input int FastEMA = 9; input int SlowEMA = 21; input int RSIPeriod = 14; input double BuyRSIMin = 52.0; input double SellRSIMax = 48.0; // SL / TP input int ATRPeriod = 14; input double StopATR = 1.8; input double TakeATR = 2.7; input double TrailATR = 1.2; input int MaxSpreadPoints = 80; input int SlippagePoints = 30; // Reverse BUY <-> SELL only when current trade is profitable input bool ReverseOnlyIfProfit = true; int hFast; int hSlow; int hRSI; int hATR; double dayStartEquity = 0.0; int dayKey = -1; //---------------------------------------------------------- // DAY //---------------------------------------------------------- int CurrentDayKey() { MqlDateTime t; TimeToStruct(TimeCurrent(), t); return t.year * 1000 + t.day_of_year; } void ResetDailyEquity() { int k = CurrentDayKey(); if(k != dayKey) { dayKey = k; dayStartEquity = AccountInfoDouble(ACCOUNT_EQUITY); } } bool DailyRiskOK() { ResetDailyEquity(); if(dayStartEquity <= 0) return true; double equity = AccountInfoDouble(ACCOUNT_EQUITY); double drawdown = 100.0 * (dayStartEquity - equity) / dayStartEquity; if(drawdown >= DailyLossLimitPct) return false; return true; } //---------------------------------------------------------- // LOT SIZE //---------------------------------------------------------- double NormalizeVolume(double lots) { double minLot = SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_MIN ); double maxLot = SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_MAX ); double step = SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_STEP ); if(step <= 0) return minLot; lots = MathMax( minLot, MathMin(maxLot, lots) ); lots = MathFloor(lots / step) * step; return NormalizeDouble(lots, 2); } double CalculateLot(double stopDistance) { if(stopDistance <= 0) return SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_MIN ); double equity = AccountInfoDouble( ACCOUNT_EQUITY ); double moneyRisk = equity * RiskPercent / 100.0; double tickSize = SymbolInfoDouble( _Symbol, SYMBOL_TRADE_TICK_SIZE ); double tickValue = SymbolInfoDouble( _Symbol, SYMBOL_TRADE_TICK_VALUE_LOSS ); if(tickSize <= 0 || tickValue <= 0) { return SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_MIN ); } double lossOneLot = (stopDistance / tickSize) * tickValue; if(lossOneLot <= 0) return SymbolInfoDouble( _Symbol, SYMBOL_VOLUME_MIN ); double lots = moneyRisk / lossOneLot; return NormalizeVolume(lots); } //---------------------------------------------------------- // SPREAD FILTER //---------------------------------------------------------- bool SpreadOK() { MqlTick tick; if(!SymbolInfoTick( _Symbol, tick )) return false; double spread = (tick.ask - tick.bid) / _Point; if(spread > MaxSpreadPoints) return false; return true; } //---------------------------------------------------------- // INDICATORS //---------------------------------------------------------- bool GetIndicators( double &fastNow, double &fastOld, double &slowNow, double &slowOld, double &rsi, double &atr ) { double fast[2]; double slow[2]; double r[1]; double a[1]; if(CopyBuffer( hFast, 0, 0, 2, fast ) != 2) return false; if(CopyBuffer( hSlow, 0, 0, 2, slow ) != 2) return false; if(CopyBuffer( hRSI, 0, 0, 1, r ) != 1) return false; if(CopyBuffer( hATR, 0, 0, 1, a ) != 1) return false; fastOld = fast[0]; fastNow = fast[1]; slowOld = slow[0]; slowNow = slow[1]; rsi = r[0]; atr = a[0]; return true; } //---------------------------------------------------------- // CANDLE + TREND SIGNAL //---------------------------------------------------------- int GetSignal( double fastNow, double fastOld, double slowNow, double slowOld, double rsi ) { MqlRates candle[3]; if(CopyRates( _Symbol, SignalTF, 0, 3, candle ) != 3) return 0; double currentBody = MathAbs( candle[2].close - candle[2].open ); double previousBody = MathAbs( candle[1].close - candle[1].open ); bool momentum = currentBody >= previousBody * 0.70; // BUY CONDITIONS bool buySignal = fastNow > slowNow && fastNow >= fastOld && rsi >= BuyRSIMin && candle[2].close > candle[2].open && momentum; // SELL CONDITIONS bool sellSignal = fastNow < slowNow && fastNow <= fastOld && rsi <= SellRSIMax && candle[2].close < candle[2].open && momentum; if(buySignal) return 1; if(sellSignal) return -1; return 0; } //---------------------------------------------------------- // TRAILING PROFIT //---------------------------------------------------------- void ManageTrailing(double atr) { if(!PositionSelect(_Symbol)) return; if( (ulong) PositionGetInteger( POSITION_MAGIC ) != MagicNumber ) return; long type = PositionGetInteger( POSITION_TYPE ); double openPrice = PositionGetDouble( POSITION_PRICE_OPEN ); double currentSL = PositionGetDouble( POSITION_SL ); double currentTP = PositionGetDouble( POSITION_TP ); MqlTick tick; if(!SymbolInfoTick( _Symbol, tick )) return; // BUY PROFIT PROTECTION if( type == POSITION_TYPE_BUY && tick.bid > openPrice ) { double newSL = NormalizeDouble( tick.bid - TrailATR * atr, _Digits ); if( newSL > openPrice && ( currentSL == 0 || newSL > currentSL ) ) { trade.PositionModify( _Symbol, newSL, currentTP ); } } // SELL PROFIT PROTECTION if( type == POSITION_TYPE_SELL && tick.ask < openPrice ) { double newSL = NormalizeDouble( tick.ask + TrailATR * atr, _Digits ); if( newSL < openPrice && ( currentSL == 0 || newSL < currentSL ) ) { trade.PositionModify( _Symbol, newSL, currentTP ); } } } //---------------------------------------------------------- // OPEN BUY / SELL //---------------------------------------------------------- bool OpenTrade( int signal, double atr ) { if(!DailyRiskOK()) return false; if(!SpreadOK()) return false; if(atr <= 0) return false; MqlTick tick; if(!SymbolInfoTick( _Symbol, tick )) return false; double stopDistance = StopATR * atr; double takeDistance = TakeATR * atr; double lot = CalculateLot( stopDistance ); // BUY if(signal > 0) { double sl = NormalizeDouble( tick.ask - stopDistance, _Digits ); double tp = NormalizeDouble( tick.ask + takeDistance, _Digits ); return trade.Buy( lot, _Symbol, 0.0, sl, tp, "DAN GOLD BUY" ); } // SELL if(signal < 0) { double sl = NormalizeDouble( tick.bid + stopDistance, _Digits ); double tp = NormalizeDouble( tick.bid - takeDistance, _Digits ); return trade.Sell( lot, _Symbol, 0.0, sl, tp, "DAN GOLD SELL" ); } return false; } //---------------------------------------------------------- // BUY -> SELL // SELL -> BUY //---------------------------------------------------------- void HandlePosition( int signal, double atr ) { // No position: // open new BUY or SELL if(!PositionSelect(_Symbol)) { if(signal != 0) OpenTrade( signal, atr ); return; } if( (ulong) PositionGetInteger( POSITION_MAGIC ) != MagicNumber ) return; long type = PositionGetInteger( POSITION_TYPE ); double profit = PositionGetDouble( POSITION_PROFIT ); bool oppositeSignal = ( type == POSITION_TYPE_BUY && signal < 0 ) || ( type == POSITION_TYPE_SELL && signal > 0 ); if(oppositeSignal) { if( ReverseOnlyIfProfit && profit <= 0 ) return; // Close current trade if( trade.PositionClose( _Symbol ) ) { // Immediately open // opposite direction OpenTrade( signal, atr ); } } } //---------------------------------------------------------- // START EA //---------------------------------------------------------- int OnInit() { trade.SetExpertMagicNumber( MagicNumber ); trade.SetDeviationInPoints( SlippagePoints ); trade.SetTypeFillingBySymbol( _Symbol ); hFast = iMA( _Symbol, SignalTF, FastEMA, 0, MODE_EMA, PRICE_CLOSE ); hSlow = iMA( _Symbol, SignalTF, SlowEMA, 0, MODE_EMA, PRICE_CLOSE ); hRSI = iRSI( _Symbol, SignalTF, RSIPeriod, PRICE_CLOSE ); hATR = iATR( _Symbol, SignalTF, ATRPeriod ); if( hFast == INVALID_HANDLE || hSlow == INVALID_HANDLE || hRSI == INVALID_HANDLE || hATR == INVALID_HANDLE ) return INIT_FAILED; ResetDailyEquity(); return INIT_SUCCEEDED; } //---------------------------------------------------------- // EVERY MARKET TICK //---------------------------------------------------------- void OnTick() { double fastNow; double fastOld; double slowNow; double slowOld; double rsi; double atr; if( !GetIndicators( fastNow, fastOld, slowNow, slowOld, rsi, atr ) ) return; // Protect existing profit ManageTrailing(atr); // Analyse current market int signal = GetSignal( fastNow, fastOld, slowNow, slowOld, rsi ); // Trade / reverse HandlePosition( signal, atr ); } //---------------------------------------------------------- // STOP EA //---------------------------------------------------------- void OnDeinit( const int reason ) { IndicatorRelease(hFast); IndicatorRelease(hSlow); IndicatorRelease(hRSI); IndicatorRelease(hATR); }
